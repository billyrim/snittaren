cmake_minimum_required(VERSION 3.22)
project(Snittaren VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum macOS")
set(JUCE_DIR "$ENV{HOME}/Downloads/JUCE-master" CACHE PATH "JUCE source directory")
add_subdirectory("${JUCE_DIR}" JUCE)
juce_add_plugin(Snittaren COMPANY_NAME "Fronas" BUNDLE_ID "se.billyrim.snittaren" PLUGIN_MANUFACTURER_CODE BlRm PLUGIN_CODE Snit FORMATS AU PRODUCT_NAME "SNITTAREN" COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(Snittaren PRIVATE Source/Plugin.cpp)
target_compile_definitions(Snittaren PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Snittaren PUBLIC juce::juce_audio_utils juce::juce_recommended_config_flags)
add_executable(dsp_tests Tests/DspTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_executable(host_tests)
target_sources(host_tests PRIVATE Tests/HostTests.cpp)
target_link_libraries(host_tests PRIVATE Snittaren)
enable_testing()
add_test(NAME dsp COMMAND dsp_tests)
add_test(NAME state COMMAND host_tests)
add_executable(render_pad)
target_sources(render_pad PRIVATE Tests/Render.cpp)
target_link_libraries(render_pad PRIVATE Snittaren)

if(APPLE)
 add_custom_command(TARGET Snittaren_AU POST_BUILD COMMAND codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:Snittaren_AU>" VERBATIM)
endif()
add_executable(compatibility_tests Tests/Compatibility.cpp)
target_include_directories(compatibility_tests PRIVATE Source)
add_test(NAME compatibility COMMAND compatibility_tests)
